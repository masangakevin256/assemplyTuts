Basic Assembly Examples

1 Hello World Program
WriteString

The classic "Hello World" program in Assembly language using Irvine32 library.

hello.asm
; Printing hello world using Irvine32 library

INCLUDE Irvine32.inc

.data
    msg BYTE "Hello world!", 0

.code
    main PROC
        call Clrscr ; clear screen, optional
        mov edx, OFFSET msg ; address of string
        call WriteString ; Irvine32 to print a zero terminating string
        call Crlf ; print new line optionally
        exit ; return to OS

    main ENDP

END main
Explanation:
  • INCLUDE Irvine32.inc includes the Irvine32 library
  • .data section defines variables and constants
  • BYTE declares a byte-sized variable
  • WriteString prints null-terminated strings
  • Crlf creates a new line (carriage return + line feed)
2 Multi-line Text Output
WriteString

Displaying multiple lines of text using carriage return and line feed characters.

paragraph.asm
INCLUDE Irvine32.inc

.data
    msg BYTE "Hello, World!", 0Dh, 0Ah, ; 0Dh is carriage return and 0Ah is Line Feed
             "My name is Kevin.", 0Dh, 0Ah,
             "I am learning Assembly.", 0

.code
    main PROC
        call Clrscr
        mov edx, OFFSET msg
        call WriteString
        exit
    main ENDP

END main
Explanation:
  • 0Dh is the ASCII code for Carriage Return
  • 0Ah is the ASCII code for Line Feed
  • Together they create a new line in Windows
  • Multiple strings can be concatenated in the data section

User Input Examples

1 Reading Integer Input
ReadInt

Read an integer from the user and display it back.

readint.asm
INCLUDE Irvine32.inc

.data
    prompt BYTE "Enter an integer: ", 0
    result BYTE "You entered: ", 0
    num DWORD ?

.code
    main PROC
        call Clrscr

        ; Display prompt
        mov edx, OFFSET prompt
        call WriteString

        ; Read integer input
        call ReadInt
        mov num, eax

        ; Display result
        mov edx, OFFSET result
        call WriteString
        mov eax, num
        call WriteInt
        call Crlf

        exit
    main ENDP

END main
Explanation:
  • ReadInt reads a 32-bit signed integer from standard input
  • The input value is stored in the EAX register
  • WriteInt displays a signed integer value
  • User input is stored in a variable for later use
2 Reading String Input
ReadString

Read a string from the user and display it back.

readstr.asm
INCLUDE Irvine32.inc

.data
    prompt BYTE "Enter your name: ", 0
    result BYTE "Hello, ", 0
    buffer BYTE 50 DUP(0) ; 50-byte buffer
    byteCount DWORD ?

.code
    main PROC
        call Clrscr

        ; Display prompt
        mov edx, OFFSET prompt
        call WriteString

        ; Read string input
        mov edx, OFFSET buffer
        mov ecx, SIZEOF buffer
        call ReadString
        mov byteCount, eax

        ; Display result
        mov edx, OFFSET result
        call WriteString
        mov edx, OFFSET buffer
        call WriteString
        call Crlf

        exit
    main ENDP

END main
Explanation:
  • DUP(0) creates an array filled with zeros
  • ReadString requires buffer address in EDX and buffer size in ECX
  • The number of characters read is returned in EAX
  • Buffer must be large enough to hold the input plus null terminator

Arithmetic Operations

1 Adding Two Numbers
ADD

Add two numbers and display the result.

addition.asm
INCLUDE Irvine32.inc

.data
    num1 DWORD 5
    num2 DWORD 3
    msg BYTE "The sum is: ", 0

.code
    main PROC
        call Clrscr

        mov eax, num1 ; eax = 5
        add eax, num2 ; eax = 5 + 3 = 8

        mov edx, OFFSET msg
        call WriteString ; print message

        call WriteInt ; print value in EAX

        call Crlf
        exit
    main ENDP

END main
Explanation:
  • ADD instruction adds two operands
  • The result is stored in the first operand
  • DWORD declares a 32-bit (double word) variable
  • Arithmetic operations typically use the EAX register
2 Multiplication with User Input
MUL

Get two numbers from user, multiply them, and display the result.

multiplication.asm
INCLUDE Irvine32.inc

.data
    prompt1 BYTE "Enter first number: ", 0
    prompt2 BYTE "Enter second number: ", 0
    resultMsg BYTE "The product is: ", 0
    num1 DWORD ?
    num2 DWORD ?

.code
    main PROC
        call Clrscr

        ; Get first number
        mov edx, OFFSET prompt1
        call WriteString
        call ReadInt
        mov num1, eax

        ; Get second number
        mov edx, OFFSET prompt2
        call WriteString
        call ReadInt
        mov num2, eax

        ; Multiply numbers
        mov eax, num1
        mul num2 ; EAX = EAX * num2

        ; Display result
        mov edx, OFFSET resultMsg
        call WriteString
        call WriteInt
        call Crlf

        exit
    main ENDP

END main
Explanation:
  • MUL multiplies EAX by the operand
  • For 32-bit multiplication, result is stored in EDX:EAX (64-bit)
  • If the result fits in 32 bits, EDX will be zero
  • Always clear the screen at the beginning for clean output

Control Flow Examples

1 If-Else Implementation
CMP/Jump

Check if a number is even or odd using conditional jumps.

conditional.asm
INCLUDE Irvine32.inc

.data
    prompt BYTE "Enter a number: ", 0
    evenMsg BYTE "The number is even.", 0
    oddMsg BYTE "The number is odd.", 0
    num DWORD ?

.code
    main PROC
        call Clrscr

        ; Get number from user
        mov edx, OFFSET prompt
        call WriteString
        call ReadInt
        mov num, eax

        ; Check if number is even or odd
        mov ebx, 2
        xor edx, edx ; clear EDX for division
        div ebx ; divide EAX by 2

        ; If remainder (EDX) is 0, number is even
        cmp edx, 0
        je isEven ; jump if equal to zero

        ; Number is odd
        mov edx, OFFSET oddMsg
        jmp displayResult

        isEven:
            mov edx, OFFSET evenMsg

        displayResult:
            call WriteString
            call Crlf

        exit
    main ENDP

END main
Explanation:
  • CMP compares two values and sets flags
  • JE jumps if the previous comparison was equal
  • JMP performs an unconditional jump
  • Division uses EDX:EAX as the dividend, result in EAX, remainder in EDX
